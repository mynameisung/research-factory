'==========================================================================
' frmSettings — 설정 팝업
'
' 리본에 입력 칸을 늘어놓으면 폭을 많이 먹고, 무엇을 바꿨는지 한눈에 들어오지
' 않는다. 기준을 한 자리에 모아 놓고 [저장] 을 눌러야 반영되게 했다.
'
' 이 모듈은 build.ps1 이 폼을 만들 때 함께 넣는다. 고칠 곳은
' src/frmSettings.code.txt 다.
'==========================================================================
Option Explicit

Private Sub UserForm_Initialize()
    txtKeep.Text = modConfig.KeepGroups()
    txtBase.Text = CStr(modConfig.MinBase())

    cboMode.Clear
    cboMode.AddItem "표준편차 (배수)"
    cboMode.AddItem "절대값 (%p)"
    If modConfig.InsightMode() = "ABS" Then
        cboMode.ListIndex = 1
    Else
        cboMode.ListIndex = 0
    End If

    txtVal.Text = CStr(modConfig.InsightValue())
    chkBlank.Value = modConfig.BlankIsZero()
    chkDisp.Value = modConfig.UseDisplayValue()
    chkBackup.Value = modConfig.BackupBeforeDelete()
    ShowHint
    ShowWhere
End Sub

' 지금 어느 시트를 놓고 설정하는지, 그 시트에서 표를 몇 장 찾았는지 알린다.
' 통계표가 아닌 시트에서 눌렀을 때 곧바로 알아채라는 것이다.
Private Sub ShowWhere()
    Dim ws As Worksheet, tabs() As TTable, n As Long
    On Error Resume Next
    Set ws = ActiveSheet
    If ws Is Nothing Then Exit Sub
    tabs = modTable.ScanTables(ws, n)
    If n > 0 Then
        Me.Caption = "통계표 전처리 - 설정      현재 시트 '" & ws.Name & "' 에서 표 " & n & "장"
    Else
        Me.Caption = "통계표 전처리 - 설정      현재 시트 '" & ws.Name & "' 에는 통계표가 없습니다"
    End If
End Sub

' 방식을 바꾸면 기준값의 뜻이 달라진다. 배수와 %p 를 헷갈리지 않도록 알린다.
Private Sub cboMode_Change()
    ShowHint
End Sub

Private Sub ShowHint()
    If cboMode.ListIndex = 1 Then
        lblHint.Caption = "절대값 : 전체 값과의 차이가 이 %p 를 넘으면 칠합니다. " & _
                          "기준이 눈에 보여 남에게 설명하기 쉽습니다. 보통 10 안팎을 씁니다."
    Else
        lblHint.Caption = "표준편차 : 그 행의 배너 값들이 흩어진 정도의 몇 배까지 벗어났는지로 " & _
                          "봅니다. 표마다 흩어진 정도가 달라도 한 기준으로 훑을 수 있습니다. 보통 2 를 씁니다."
    End If
End Sub

Private Sub btnSave_Click()
    Dim b As Double, v As Double

    If Len(Trim$(txtBase.Text)) = 0 Or Not IsNumeric(txtBase.Text) Then
        MsgBox "Base 하한에 숫자를 넣어 주세요. 통계표에서 % 를 인용할 수 있는 " & _
               "최소 모수는 30 입니다.", vbExclamation, "설정"
        txtBase.SetFocus
        Exit Sub
    End If
    b = Val(txtBase.Text)
    If b < 1 Then
        MsgBox "Base 하한은 1 이상이어야 합니다.", vbExclamation, "설정"
        txtBase.SetFocus
        Exit Sub
    End If
    If b < 30 Then
        If MsgBox("Base 하한을 " & b & " 로 두면 모수가 모자란 배너까지 남습니다." & vbCrLf & _
                  "그 칸의 % 는 튀는 값이 아니라 표본 잡음일 수 있습니다." & vbCrLf & vbCrLf & _
                  "그래도 이 값으로 하시겠습니까?", vbYesNo + vbExclamation, "설정") <> vbYes Then
            txtBase.SetFocus
            Exit Sub
        End If
    End If

    If Len(Trim$(txtVal.Text)) = 0 Or Not IsNumeric(txtVal.Text) Then
        MsgBox "인사이트 기준값에 숫자를 넣어 주세요.", vbExclamation, "설정"
        txtVal.SetFocus
        Exit Sub
    End If
    v = Val(txtVal.Text)
    If v <= 0 Then
        MsgBox "인사이트 기준값은 0 보다 커야 합니다.", vbExclamation, "설정"
        txtVal.SetFocus
        Exit Sub
    End If

    modConfig.SetKeepGroups Trim$(txtKeep.Text)
    modConfig.SetMinBase CStr(b)
    modConfig.SetInsightMode IIf(cboMode.ListIndex = 1, "ABS", "STDEV")
    modConfig.SetInsightValue CStr(v)
    modConfig.SetBlankIsZero CBool(chkBlank.Value)
    modConfig.SetUseDisplayValue CBool(chkDisp.Value)
    modConfig.SetBackupBeforeDelete CBool(chkBackup.Value)

    Unload Me
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub

Private Sub btnReset_Click()
    If MsgBox("기준을 처음 상태로 되돌립니다." & vbCrLf & vbCrLf & _
              "보존 배너군은 비우고, Base 하한 30 · 표준편차 2배 ·" & vbCrLf & _
              "세 규약은 모두 켬으로 돌아갑니다." & vbCrLf & vbCrLf & _
              "계속하시겠습니까?", vbYesNo + vbExclamation, "설정 초기화") <> vbYes Then Exit Sub
    txtKeep.Text = ""
    txtBase.Text = "30"
    cboMode.ListIndex = 0
    txtVal.Text = "2"
    chkBlank.Value = False
    chkDisp.Value = True
    chkBackup.Value = True
    ShowHint
End Sub
