'==========================================================================
' frmCatalogPick — [배너군 목록] 을 뽑을 문항을 고르는 팝업
'
' 보통은 시트의 문항 전체를 기준으로 배너군을 뽑지만, 문항마다 배너 구성이
' 다를 수 있어 특정 문항 하나만의 배너군을 보고 싶을 때가 있다. 그래서
' [배너군 목록] 을 누르면 이 팝업이 먼저 뜨고, 기본은 전체 선택이다.
'
' 이 모듈은 build.ps1 이 폼을 만들 때 함께 넣는다. 고칠 곳은
' src/frmCatalogPick.code.txt 다.
'==========================================================================
Option Explicit

Private mCount As Long

' modReport.BannerCatalog 이 스캔해 둔 문항 제목을 여기로 채워 준다.
' titles 는 1..n 이다(표 번호와 같은 번호를 그대로 쓴다).
Public Sub Populate(titles() As String)
    Dim lo As Long, hi As Long, i As Long
    lo = LBound(titles): hi = UBound(titles)
    mCount = hi - lo + 1

    lstQ.Clear
    For i = lo To hi
        lstQ.AddItem CStr(i) & ".  " & titles(i)
        lstQ.Selected(lstQ.ListCount - 1) = True   ' 기본값 - 전체 선택
    Next i

    lblInfo.Caption = "문항 " & mCount & "개 가운데 배너군을 뽑을 문항을 고르세요." & vbCrLf & _
                      "보통은 전체 선택 그대로 두고, 특정 문항 하나만의 배너군 구성이" & vbCrLf & _
                      "궁금할 때만 좁혀서 고르세요."
    Me.Caption = "통계표 전처리 - 배너군 목록  (문항 " & mCount & "개)"
End Sub

Private Sub btnAll_Click()
    Dim i As Long
    For i = 0 To lstQ.ListCount - 1
        lstQ.Selected(i) = True
    Next i
End Sub

Private Sub btnNone_Click()
    Dim i As Long
    For i = 0 To lstQ.ListCount - 1
        lstQ.Selected(i) = False
    Next i
End Sub

Private Sub btnOK_Click()
    Dim i As Long, any As Boolean
    Dim picked() As Boolean
    ReDim picked(1 To mCount)
    For i = 0 To lstQ.ListCount - 1
        picked(i + 1) = lstQ.Selected(i)
        If picked(i + 1) Then any = True
    Next i
    If Not any Then
        MsgBox "적어도 문항 하나는 고르셔야 합니다.", vbExclamation, "배너군 목록"
        Exit Sub
    End If
    Unload Me
    modReport.RunCatalogFiltered picked
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub
